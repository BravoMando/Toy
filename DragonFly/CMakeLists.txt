file(GLOB_RECURSE SOURCE_C_CPP
  ${CMAKE_CURRENT_SOURCE_DIR}/public/*.c
  ${CMAKE_CURRENT_SOURCE_DIR}/public/*.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/private/*.c
  ${CMAKE_CURRENT_SOURCE_DIR}/private/*.cpp
)
file(GLOB_RECURSE MODULE_INTERFACE
  ${CMAKE_CURRENT_SOURCE_DIR}/public/*.cppm
)
file(GLOB_RECURSE HEADER_INTERFACE
  ${CMAKE_CURRENT_SOURCE_DIR}/public/*.h
)

add_library(${DRAGONFLY_NAME} SHARED)
target_sources(${DRAGONFLY_NAME}
  PRIVATE
  ${SOURCE_C_CPP}
  PUBLIC
  FILE_SET CXX_MODULES
  BASE_DIRS ${CMAKE_CURRENT_SOURCE_DIR}/public
  FILES ${MODULE_INTERFACE}
  FILE_SET HEADERS
  BASE_DIRS ${CMAKE_CURRENT_SOURCE_DIR}/public
  FILES ${HEADER_INTERFACE}
)
target_include_directories(${DRAGONFLY_NAME} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/public)
target_compile_definitions(${DRAGONFLY_NAME} PRIVATE BUILD_SDK=1)
set_target_properties(${DRAGONFLY_NAME} PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(${DRAGONFLY_NAME} PRIVATE $<$<CONFIG:DEBUG>:-O0>
  $<$<CONFIG:DEBUG>:-g> $<$<CONFIG:RELEASE>:-O2>)
